```swift
import Foundation

public struct VPNConfig: Identifiable, Hashable, Sendable {
    public let id: UUID
    public let name: String
    public let fileURL: URL

    public init(fileURL: URL) {
        self.id = UUID()
        self.fileURL = fileURL
        self.name = fileURL.deletingPathExtension().lastPathComponent
    }
}
```
All members MUST be `public` — App and CLI targets import VPNCore across module boundary.

**File 2: VPNCore/ConnectionState.swift**

Use `case failed(String)` storing the error message for clean `Sendable` conformance under Swift 6 strict concurrency:
```swift
import Foundation

public enum ConnectionState: Sendable {
    case disconnected
    case authenticating
    case connected
    case disconnecting
    case failed(String)

    public var isConnected: Bool {
        if case .connected = self { return true }
        return false
    }
}
```
Do NOT conform to Equatable — the `isConnected` computed property is sufficient for Phase 1.

**File 3: VPNCore/VPNManager.swift**
```swift
import Foundation
import Observation

@Observable
@MainActor
public final class VPNManager {
    public var configs: [VPNConfig] = []
    public var connections: [String: ConnectionState] = [:]

    public var isAnyConnected: Bool {
        connections.values.contains { $0.isConnected }
    }

    public static let configsDirectory: URL = {
        FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
            .appendingPathComponent("AWSVPNClient/configs", isDirectory: true)
    }()

    public init() {
        try? FileManager.default.createDirectory(
            at: Self.configsDirectory,
            withIntermediateDirectories: true,
            attributes: nil
        )
    }

    public func connect(_ config: VPNConfig) async throws {
        fatalError("Phase 2")
    }

    public func disconnect(_ config: VPNConfig) async throws {
        fatalError("Phase 2")
    }
}
```

Key details:
- Every type, property, method, and init MUST be `public`
- `@Observable` from Observation framework (NOT ObservableObject)
- `@MainActor` on the class for safe main-thread mutation
- `configsDirectory` is a static computed URL using FileManager
- `init()` calls `createDirectory(withIntermediateDirectories: true)` — idempotent, safe every launch
- `connect()` and `disconnect()` use `fatalError("Phase 2")` — Phase 2 fills these in
- `isAnyConnected` uses the `isConnected` computed property on ConnectionState

**File 1: AWSVPNClient/AWSVPNClientApp.swift** (full replacement)
```swift
import SwiftUI
import VPNCore

@main
@MainActor
struct AWSVPNClientApp: App {
    @State private var vpnManager = VPNManager()

    var body: some Scene {
        MenuBarExtra(
            vpnManager.isAnyConnected ? "VPN Connected" : "VPN",
            systemImage: vpnManager.isAnyConnected ? "lock.fill" : "lock.open"
        ) {
            StatusMenuView()
                .environment(vpnManager)
        }
        .menuBarExtraStyle(.menu)
    }
}
```

Key details:
- `@MainActor` on the App struct — required to resolve Swift 6 error when `@State` initializes a `@MainActor`-isolated class (Pitfall A)
- `@State private var vpnManager = VPNManager()` — NOT `@StateObject` (Pitfall: @StateObject is for ObservableObject, not @Observable)
- `.environment(vpnManager)` passes VPNManager to child views via @Environment
- Icon: `lock.fill` when `isAnyConnected` is true, `lock.open` otherwise — per locked CONTEXT.md decision
- `.menuBarExtraStyle(.menu)` — native pull-down menu style per locked decision

**File 2: AWSVPNClient/StatusMenuView.swift** (new file)

```swift
import SwiftUI
import VPNCore

struct StatusMenuView: View {
    @Environment(VPNManager.self) private var vpnManager

    var body: some View {
        if vpnManager.configs.isEmpty {
            Text("No configs — add .conf files")
                .foregroundStyle(.secondary)
        }

        Divider()

        Button("Quit AWSVPNClient") {
            NSApplication.shared.terminate(nil)
        }
    }
}
```

Key details:
- `@Environment(VPNManager.self)` — the Observation-framework way to read @Observable objects (NOT `@EnvironmentObject`)

- `NSApplication.shared.terminate(nil)` triggers standard app termination including applicationWillTerminate
- Empty config state shows informational text (Phase 3 adds the add/remove config UI)
