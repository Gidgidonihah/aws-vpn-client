```swift
import Foundation

public enum VPNConfigParser {
    public struct ParsedConfig: Sendable {
        public let host: String
        public let port: String
        public let proto: String
        public let filteredContent: String
    }

    /// Parse a .conf file content string. Extracts host/port/proto from "remote" and "proto"
    /// directives, and returns filtered content with auth/remote directives stripped.
    public static func parse(content: String) throws -> ParsedConfig {
        let lines = content.components(separatedBy: "\n")

        guard let host = field(lines: lines, directive: "remote", index: 1) else {
            throw VPNError.configParseFailure("No 'remote' directive found")
        }
        guard let port = field(lines: lines, directive: "remote", index: 2) else {
            throw VPNError.configParseFailure("No port in 'remote' directive")
        }
        guard let proto = field(lines: lines, directive: "proto", index: 1) else {
            throw VPNError.configParseFailure("No 'proto' directive found")
        }

        let filtered = lines
            .filter { !shouldStripLine($0) }
            .joined(separator: "\n")

        return ParsedConfig(host: host, port: port, proto: proto, filteredContent: filtered)
    }

    /// Convenience: parse from a file URL
    public static func parse(fileURL: URL) throws -> ParsedConfig {
        let content = try String(contentsOf: fileURL, encoding: .utf8)
        return try parse(content: content)
    }

    private static func shouldStripLine(_ line: String) -> Bool {
        let l = line.trimmingCharacters(in: .whitespaces)
        return l.hasPrefix("auth-user-pass")
            || l.hasPrefix("auth-federate")
            || (l.hasPrefix("auth-retry") && l.contains("interact"))
            || l.hasPrefix("remote")
    }

    private static func field(lines: [String], directive: String, index: Int) -> String? {
        let prefix = directive + " "
        guard let line = lines.first(where: {
            $0.trimmingCharacters(in: .whitespaces).hasPrefix(prefix)
        }) else { return nil }
        let parts = line.split(separator: " ").map(String.init)
        guard parts.count > index else { return nil }
        return parts[index]
    }
}
```

```swift
import Foundation
import Security

/// Generates `byteCount` random bytes as a lowercase hex string.
/// randomHex(byteCount: 12) produces a 24-character hex string.
public func randomHex(byteCount: Int) -> String {
    var bytes = [UInt8](repeating: 0, count: byteCount)
    _ = SecRandomCopyBytes(kSecRandomDefault, byteCount, &bytes)
    return bytes.map { String(format: "%02x", $0) }.joined()
}

/// URL-encodes a SAML response string for openvpn credentials.
/// Uses a custom character set that excludes '+' (unlike .urlQueryAllowed).
/// Matches Rust's urlencoding::encode behavior: space -> %20, + -> %2B.
public func urlEncodeSAML(_ samlResponse: String) -> String {
    var allowed = CharacterSet.urlQueryAllowed
    allowed.remove("+")
    return samlResponse.addingPercentEncoding(withAllowedCharacters: allowed) ?? samlResponse
}

/// Dummy credentials for the initial openvpn auth challenge.
/// Line 1: "N/A", Line 2: "ACS::35001"
public func dummyCredentials() -> String {
    "N/A\nACS::35001\n"
}

/// Real credentials with the SAML token for the sudo openvpn connection.
/// Line 1: "N/A", Line 2: "CRV1::<sid>::<urlEncodedSAML>"
public func realCredentials(sid: String, urlEncodedSAML: String) -> String {
    "N/A\nCRV1::\(sid)::\(urlEncodedSAML)\n"
}

/// Result of parsing a CRV1 challenge line from openvpn output.
public struct CRV1Challenge: Sendable {
    public let url: String
    public let sid: String
}

/// Parses an AUTH_FAILED,CRV1 line from openvpn stdout/stderr.
/// Extracts the SAML redirect URL and session ID.
/// Returns nil if the line does not contain "AUTH_FAILED,CRV1".
public func parseCRV1Line(_ line: String) -> CRV1Challenge? {
    guard line.contains("AUTH_FAILED,CRV1") else { return nil }

    // Extract URL: everything after "https://"
    guard let urlPart = line.split(separator: "https://", maxSplits: 1).last else { return nil }
    let url = "https://" + urlPart.trimmingCharacters(in: .whitespacesAndNewlines)

    // Extract SID: colon-split field at zero-based index 6
    let parts = line.split(separator: ":", omittingEmptySubsequences: false).map(String.init)
    guard parts.count > 6 else { return nil }
    let sid = parts[6]
    guard !sid.isEmpty else { return nil }

    return CRV1Challenge(url: url, sid: sid)
}

/// Finds the first line containing "AUTH_FAILED,CRV1" in combined openvpn output.
public func findCRV1Line(in output: String) -> String? {
    output.components(separatedBy: "\n")
        .first { $0.contains("AUTH_FAILED,CRV1") }
}
```

IMPORTANT: `line.split(separator: "https://", ...)` — in Swift, `String.split(separator:)` with a String separator works differently than Rust. Use `components(separatedBy: "https://")` and take `dropFirst().first` instead, matching the Rust pattern `split("https://").nth(1)`:

```swift
guard let urlSuffix = line.components(separatedBy: "https://").dropFirst().first else { return nil }
let url = "https://" + urlSuffix.trimmingCharacters(in: .whitespacesAndNewlines)
```

For colon-split, use `line.components(separatedBy: ":")` (NOT `split(separator:)` which omits empty subsequences by default):

```swift
let parts = line.components(separatedBy: ":")
guard parts.count > 6 else { return nil }
let sid = parts[6]
```
